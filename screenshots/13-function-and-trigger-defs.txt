--- FUNCTION: log_contract_status_change ---
CREATE OR REPLACE FUNCTION public.log_contract_status_change()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
  IF NEW.status IS DISTINCT FROM OLD.status THEN
    INSERT INTO audit_log (user_id, action, entity, entity_id)
    VALUES (
      NULLIF(current_setting('app.user_id', true), '')::int,
      'STATUS ' || OLD.status || ' -> ' || NEW.status,
      'contract',
      NEW.contract_id
    );
  END IF;
  RETURN NEW;
END;
$function$


--- FUNCTION: get_contract_summary ---
CREATE OR REPLACE FUNCTION public.get_contract_summary(p_contract_id integer)
 RETURNS TABLE(total_milestones integer, completed_milestones integer, total_value numeric, completed_value numeric)
 LANGUAGE plpgsql
AS $function$
BEGIN
  RETURN QUERY
  SELECT 
    COUNT(*)::INT AS total_milestones,
    COUNT(CASE WHEN status = 'COMPLETED' THEN 1 END)::INT AS completed_milestones,
    COALESCE(SUM(amount), 0.00) AS total_value,
    COALESCE(SUM(CASE WHEN status = 'COMPLETED' THEN amount ELSE 0 END), 0.00) AS completed_value
  FROM milestone
  WHERE contract_id = p_contract_id;
END;
$function$


--- FUNCTION: find_similar_standard_clauses ---
CREATE OR REPLACE FUNCTION public.find_similar_standard_clauses(query_embedding vector, p_clause_type character varying, match_count integer DEFAULT 3)
 RETURNS TABLE(standard_clause_id integer, clause_type character varying, clause_text text, similarity double precision)
 LANGUAGE sql
 STABLE
AS $function$
  SELECT sc.standard_clause_id,
         sc.clause_type,
         sc.clause_text,
         1 - (sc.embedding <=> query_embedding) AS similarity
  FROM standard_clause sc
  WHERE sc.clause_type = p_clause_type
  ORDER BY sc.embedding <=> query_embedding
  LIMIT match_count;
$function$


--- FUNCTION: check_milestone_date_range ---
CREATE OR REPLACE FUNCTION public.check_milestone_date_range()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
  v_start_date DATE;
  v_end_date DATE;
BEGIN
  SELECT start_date, end_date INTO v_start_date, v_end_date
  FROM contract
  WHERE contract_id = NEW.contract_id;

  IF v_start_date IS NULL OR v_end_date IS NULL THEN
    RETURN NEW;
  END IF;

  IF NEW.due_date < v_start_date OR NEW.due_date > v_end_date THEN
    RAISE EXCEPTION 'Milestone due_date (%) must fall within the contract validity period (% to %)',
      NEW.due_date, v_start_date, v_end_date;
  END IF;

  RETURN NEW;
END;
$function$
